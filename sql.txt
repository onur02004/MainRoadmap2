CREATE TABLE custom_themes (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id VARCHAR(50) NOT NULL, -- örn: 'onur' veya session ID
  theme_key VARCHAR(50) UNIQUE NOT NULL,
  name VARCHAR(100) NOT NULL,
  category VARCHAR(20) DEFAULT 'custom',
  has_cli BOOLEAN DEFAULT true,
  show_grid BOOLEAN DEFAULT true,
  dot_color VARCHAR(20) NOT NULL,
  status_text VARCHAR(100),
  
  -- Özelleştirilebilir CSS Değişkenleri (JSONB)
  css_variables JSONB NOT NULL DEFAULT '{
    "--bg-color": "#0a0a0f",
    "--panel-bg": "#12121c",
    "--text-main": "#e0e0e0",
    "--text-bright": "#ffffff",
    "--accent-action": "#00ffaa",
    "--border-color": "rgba(255, 255, 255, 0.15)"
  }'::jsonb,
  
  -- Arka Plan Resmi ve Gizlilik
  bg_image_url TEXT,
  is_public BOOLEAN DEFAULT false,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TYPE visibility_type AS ENUM ('public', 'friends', 'custom');

CREATE TABLE song_suggestions (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id uuid REFERENCES users(id) ON DELETE SET NULL,
    song_name text NOT NULL,
    song_artist text NOT NULL,
    spotify_uri text,
    song_url text,
    song_url_backup text,
    song_cover_url text,
    song_cover_url_backup text,
    song_artist_cover_url text,
    song_artist_cover_url_backup text,
    song_release_date date,
    song_artist_genre text[],
    overall_dominant_color text,
    dominant_colors_points text[],
    
    -- Değişen / Yeni Alanlar
    visibility visibility_type NOT NULL DEFAULT 'public',
    target_users uuid[], -- custom veya friends seçilirse hedef kullanıcılar
    rating_by_user smallint CHECK (rating_by_user >= 0 AND rating_by_user <= 100),
    comment_by_user text,
    recommended_time_by_user character varying(10),
    song_played_by_user boolean DEFAULT false,
    
    -- Şarkıyı oynatanlar / görüntüleyenler (Kullanıcı ID listesi)
    played_or_viewed_by uuid[] DEFAULT '{}',
    
    -- Repost Edenler
    reposted_by uuid[] DEFAULT '{}',
    
    -- Lyrics (Time-synced ve Normal ayrı kolonlar veya JSON yapısı)
    lyrics text,
    time_synced_lyrics jsonb,
    
    -- Sayaçlar ve Tarihler
    likes_count integer NOT NULL DEFAULT 0,
    repost_count integer NOT NULL DEFAULT 0,
    date_added timestamp with time zone NOT NULL DEFAULT now(),
    date_edited timestamp with time zone
);

CREATE INDEX idx_song_suggestions_user_id ON song_suggestions (user_id);

CREATE TABLE IF NOT EXISTS game_catalog (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    -- Oyunun sistemdeki benzersiz kimliği
    game_key VARCHAR(100) NOT NULL UNIQUE,

    -- Görünen bilgiler
    name VARCHAR(150) NOT NULL,
    short_name VARCHAR(60),
    tagline VARCHAR(255),
    description TEXT,
    instructions TEXT,

    -- Kategori ve zorluk
    category VARCHAR(50) NOT NULL DEFAULT 'party',
    difficulty VARCHAR(30) NOT NULL DEFAULT 'easy',

    -- Oyuncu bilgileri
    min_players INTEGER NOT NULL DEFAULT 2,
    max_players INTEGER NOT NULL DEFAULT 2,
    player_modes JSONB NOT NULL DEFAULT '["multiplayer"]'::jsonb,

    -- Tahmini oyun süresi (dakika)
    estimated_duration INTEGER,

    -- Görseller
    logo_url TEXT,
    cover_image_url TEXT,
    preview_image_url TEXT,
    icon VARCHAR(100),
    accent_color VARCHAR(20) DEFAULT '#8b5cf6',

    -- Oyun sayfasının adresi
    route VARCHAR(255),

    -- Yayın ve yönetim ayarları
    status VARCHAR(20) NOT NULL DEFAULT 'draft',
    is_active BOOLEAN NOT NULL DEFAULT FALSE,
    is_featured BOOLEAN NOT NULL DEFAULT FALSE,
    sort_order INTEGER NOT NULL DEFAULT 0,

    -- Oyuna özel kurallar ve ayarlar
    rules JSONB NOT NULL DEFAULT '{}'::jsonb,
    config JSONB NOT NULL DEFAULT '{}'::jsonb,

    -- Tarihler
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    -- Veri doğrulamaları
    CONSTRAINT game_catalog_player_count_check
        CHECK (
            min_players >= 1
            AND max_players >= min_players
        ),

    CONSTRAINT game_catalog_duration_check
        CHECK (
            estimated_duration IS NULL
            OR estimated_duration > 0
        ),

    CONSTRAINT game_catalog_status_check
        CHECK (
            status IN (
                'draft',
                'active',
                'maintenance',
                'archived'
            )
        ),

    CONSTRAINT game_catalog_difficulty_check
        CHECK (
            difficulty IN (
                'easy',
                'medium',
                'hard'
            )
        )
);



CREATE INDEX IF NOT EXISTS idx_game_catalog_status
    ON game_catalog(status);

CREATE INDEX IF NOT EXISTS idx_game_catalog_category
    ON game_catalog(category);

CREATE INDEX IF NOT EXISTS idx_game_catalog_active
    ON game_catalog(is_active);

CREATE INDEX IF NOT EXISTS idx_game_catalog_sort_order
    ON game_catalog(sort_order);

CREATE INDEX IF NOT EXISTS idx_game_catalog_created_at
    ON game_catalog(created_at DESC);
