CREATE TABLE custom_themes (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id VARCHAR(50) NOT NULL, -- örn: 'onur' veya session ID
  theme_key VARCHAR(50) UNIQUE NOT NULL,
  name VARCHAR(100) NOT NULL,
  category VARCHAR(20) DEFAULT 'custom',
  has_cli BOOLEAN DEFAULT true,
  show_grid BOOLEAN DEFAULT true,
  dot_color VARCHAR(20) NOT NULL,
  status_text VARCHAR(100),
  
  -- Özelleştirilebilir CSS Değişkenleri (JSONB)
  css_variables JSONB NOT NULL DEFAULT '{
    "--bg-color": "#0a0a0f",
    "--panel-bg": "#12121c",
    "--text-main": "#e0e0e0",
    "--text-bright": "#ffffff",
    "--accent-action": "#00ffaa",
    "--border-color": "rgba(255, 255, 255, 0.15)"
  }'::jsonb,
  
  -- Arka Plan Resmi ve Gizlilik
  bg_image_url TEXT,
  is_public BOOLEAN DEFAULT false,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TYPE visibility_type AS ENUM ('public', 'friends', 'custom');

CREATE TABLE song_suggestions (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id uuid REFERENCES users(id) ON DELETE SET NULL,
    song_name text NOT NULL,
    song_artist text NOT NULL,
    spotify_uri text,
    song_url text,
    song_url_backup text,
    song_cover_url text,
    song_cover_url_backup text,
    song_artist_cover_url text,
    song_artist_cover_url_backup text,
    song_release_date date,
    song_artist_genre text[],
    overall_dominant_color text,
    dominant_colors_points text[],
    
    -- Değişen / Yeni Alanlar
    visibility visibility_type NOT NULL DEFAULT 'public',
    target_users uuid[], -- custom veya friends seçilirse hedef kullanıcılar
    rating_by_user smallint CHECK (rating_by_user >= 0 AND rating_by_user <= 100),
    comment_by_user text,
    recommended_time_by_user character varying(10),
    song_played_by_user boolean DEFAULT false,
    
    -- Şarkıyı oynatanlar / görüntüleyenler (Kullanıcı ID listesi)
    played_or_viewed_by uuid[] DEFAULT '{}',
    
    -- Repost Edenler
    reposted_by uuid[] DEFAULT '{}',
    
    -- Lyrics (Time-synced ve Normal ayrı kolonlar veya JSON yapısı)
    lyrics text,
    time_synced_lyrics jsonb,
    
    -- Sayaçlar ve Tarihler
    likes_count integer NOT NULL DEFAULT 0,
    repost_count integer NOT NULL DEFAULT 0,
    date_added timestamp with time zone NOT NULL DEFAULT now(),
    date_edited timestamp with time zone
);

CREATE INDEX idx_song_suggestions_user_id ON song_suggestions (user_id);

CREATE TABLE avatar_designs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    user_id UUID NOT NULL
        REFERENCES users(id)
        ON DELETE CASCADE,

    design_type TEXT NOT NULL,

    name TEXT NOT NULL,

    design_data JSONB NOT NULL DEFAULT '{}'::jsonb,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT avatar_designs_type_check
        CHECK (
            design_type IN (
                'pixel',
                'blobatar'
            )
        )
);


CREATE INDEX idx_avatar_designs_user_id
ON avatar_designs(user_id);

CREATE TABLE song_ratings (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),

    song_share_id uuid NOT NULL,
    user_id uuid NOT NULL,

    rating smallint NOT NULL,

    created_at timestamptz NOT NULL DEFAULT NOW(),
    updated_at timestamptz NOT NULL DEFAULT NOW(),

    CONSTRAINT song_ratings_rating_check
        CHECK (rating >= 0 AND rating <= 100),

    CONSTRAINT song_ratings_unique_user_share
        UNIQUE (song_share_id, user_id),

    CONSTRAINT song_ratings_song_share_fk
        FOREIGN KEY (song_share_id)
        REFERENCES song_suggestions(id)
        ON DELETE CASCADE,

    CONSTRAINT song_ratings_user_fk
        FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
);

CREATE INDEX idx_song_ratings_song_share_id
    ON song_ratings(song_share_id);

CREATE INDEX idx_song_ratings_user_id
    ON song_ratings(user_id);

    CREATE TABLE song_comments (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),

    song_share_id uuid NOT NULL,
    user_id uuid NOT NULL,

    comment text NOT NULL,

    created_at timestamptz NOT NULL DEFAULT NOW(),
    updated_at timestamptz NOT NULL DEFAULT NOW(),

    CONSTRAINT song_comments_song_share_fk
        FOREIGN KEY (song_share_id)
        REFERENCES song_suggestions(id)
        ON DELETE CASCADE,

    CONSTRAINT song_comments_user_fk
        FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
);

CREATE INDEX idx_song_comments_song_share_id
    ON song_comments(song_share_id);

CREATE INDEX idx_song_comments_user_id
    ON song_comments(user_id);

CREATE INDEX idx_song_comments_created_at
    ON song_comments(created_at);
